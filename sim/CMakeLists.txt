cmake_minimum_required(VERSION 3.16)
project(sim C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_PATH ${CMAKE_SOURCE_DIR}/../include/lv_conf.h CACHE STRING "" FORCE)
add_compile_definitions(LV_CONF_INCLUDE_SIMPLE)
include_directories(${CMAKE_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/../include /root/vendor/ArduinoJson/src)
add_subdirectory(${LVGL_DIR} lvgl)
target_include_directories(lvgl PUBLIC ${CMAKE_SOURCE_DIR}/../include)
file(GLOB FONTS ../src/fonts/*.c)
add_executable(sim sim_main.cpp ../src/ui.cpp ../src/dotfx.cpp ../src/xbloom_protocol.cpp ${FONTS} ../src/lv_mem_psram.c net_stub.cpp cloud_stub.cpp storage_stub.cpp ../src/library.cpp)
target_compile_definitions(sim PRIVATE UI_SIM)
set_source_files_properties(${FONTS} ../src/lv_mem_psram.c PROPERTIES COMPILE_DEFINITIONS LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(sim lvgl)
